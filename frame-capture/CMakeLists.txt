cmake_minimum_required(VERSION 3.16)
project(gamestat_frame_capture VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
include(GNUInstallDirs)

option(GAMESTAT_BUILD_VULKAN "Build the Vulkan implicit/explicit layer" ON)
option(GAMESTAT_BUILD_OPENGL "Build the OpenGL GLX/EGL preload interceptor" ON)
option(GAMESTAT_BUILD_TESTS "Build local event sender test" ON)

add_library(gamestat_event SHARED src/common/event_sender.cpp)
target_include_directories(gamestat_event PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_compile_options(gamestat_event PRIVATE -Wall -Wextra -Wpedantic)
set_target_properties(gamestat_event PROPERTIES OUTPUT_NAME gamestat_event)

if(GAMESTAT_BUILD_VULKAN)
  find_package(Vulkan QUIET)
  if(Vulkan_FOUND)
    add_library(gamestat_vulkan_layer SHARED src/vulkan/vulkan_layer.cpp)
    target_include_directories(gamestat_vulkan_layer PRIVATE include)
    target_link_libraries(gamestat_vulkan_layer PRIVATE Vulkan::Vulkan gamestat_event dl pthread)
    target_compile_options(gamestat_vulkan_layer PRIVATE -Wall -Wextra -Wpedantic)
    set_target_properties(gamestat_vulkan_layer PROPERTIES OUTPUT_NAME gamestat_vulkan_layer)
    configure_file(manifests/VkLayer_GAMestat_frame_capture.json.in VkLayer_GAMestat_frame_capture.json @ONLY)
    install(TARGETS gamestat_vulkan_layer LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR})
    install(FILES ${CMAKE_CURRENT_BINARY_DIR}/VkLayer_GAMestat_frame_capture.json DESTINATION ${CMAKE_INSTALL_DATADIR}/vulkan/implicit_layer.d)
  else()
    message(WARNING "Vulkan headers/loader not found; Vulkan layer will not be built")
  endif()
endif()

if(GAMESTAT_BUILD_OPENGL)
  find_package(X11 QUIET)
  find_library(OPENGL_GL_LIBRARY NAMES GL OpenGL)
  find_path(GLX_INCLUDE_DIR GL/glx.h)
  find_path(EGL_INCLUDE_DIR EGL/egl.h)
  find_library(EGL_LIBRARY NAMES EGL)
  if(X11_FOUND AND OPENGL_GL_LIBRARY AND GLX_INCLUDE_DIR)
    add_library(gamestat_opengl_intercept SHARED src/opengl/glx_interceptor.cpp)
    target_include_directories(gamestat_opengl_intercept PRIVATE include ${GLX_INCLUDE_DIR} ${X11_INCLUDE_DIR})
    target_link_libraries(gamestat_opengl_intercept PRIVATE gamestat_event ${CMAKE_DL_LIBS} ${OPENGL_GL_LIBRARY} pthread)
    target_compile_options(gamestat_opengl_intercept PRIVATE -Wall -Wextra -Wpedantic)
    set_target_properties(gamestat_opengl_intercept PROPERTIES OUTPUT_NAME gamestat_opengl_intercept)
    install(TARGETS gamestat_opengl_intercept LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR})
  else()
    message(WARNING "GLX development files not found; GLX interceptor will not be built")
  endif()
  if(EGL_INCLUDE_DIR AND EGL_LIBRARY)
    add_library(gamestat_egl_intercept SHARED src/opengl/egl_interceptor.cpp)
    target_include_directories(gamestat_egl_intercept PRIVATE include ${EGL_INCLUDE_DIR})
    target_link_libraries(gamestat_egl_intercept PRIVATE gamestat_event ${CMAKE_DL_LIBS} ${EGL_LIBRARY} pthread)
    target_compile_options(gamestat_egl_intercept PRIVATE -Wall -Wextra -Wpedantic)
    set_target_properties(gamestat_egl_intercept PROPERTIES OUTPUT_NAME gamestat_egl_intercept)
    install(TARGETS gamestat_egl_intercept LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR})
  else()
    message(WARNING "EGL development files not found; EGL interceptor will not be built")
  endif()
endif()

if(GAMESTAT_BUILD_TESTS)
  add_executable(gamestat_event_test examples/send_test_event.cpp)
  target_link_libraries(gamestat_event_test PRIVATE gamestat_event)
endif()

install(TARGETS gamestat_event LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
